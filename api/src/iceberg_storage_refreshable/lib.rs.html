18//! A [`Storage`] wrapper that transparently refreshes credentials and retries once on
19//! failure.
20//!
21//! This is fork-only functionality with no upstream equivalent: it exists to back the
22//! REST catalog's vended-storage-credentials feature for tables whose credentials expire
23//! while a `FileIO` built once, up front, is held and reused across many requests. It is
24//! built entirely against `iceberg`'s public `Storage`/`StorageFactory` traits (see
25//! `iceberg::io::storage::mod` for their contract) and `iceberg-storage-opendal`'s public
26//! API — it never reaches into either crate's internals, so it stays out of the way of
27//! future upstream syncs.
28//!
29//! Every [`Storage`] method call is retried at most once: on any error, credentials are
30//! refreshed via the injected [`StorageCredentialsLoader`] and the whole backend is
31//! rebuilt, then the original call is retried against the fresh backend. This is
32//! deliberately coarse (call-level, not request-level) and deliberately triggers on any
33//! error rather than trying to classify auth-shaped failures specifically — see the
34//! [`RefreshableStorage`] docs for why.
35

52/// Shared, mutable state behind a [`RefreshableStorage`]. Lives behind a single `Arc` so
53/// that cloning the outer handle (as `new_input`/`new_output` must, to hand a
54/// `Storage`-implementing value to [`InputFile`]/[`OutputFile`]) shares state rather than
55/// forking it — a refresh performed via one clone must be visible to every other clone
56/// and to every already-issued `InputFile`/`OutputFile`.
57struct State {
58    /// The current backend and the credential version it was built with, updated
59    /// together so a reader never observes a backend paired with the wrong version.
60    current: RwLock<(Arc<dyn Storage>, u64)>,
61    /// Factory used to (re)build the backend from merged base + credential properties.
62    /// Defaults to [`OpenDalResolvingStorageFactory`], which resolves the concrete
63    /// backend per call from the path's scheme rather than once at construction time —
64    /// so, unlike the old fork design, this needs no separate scheme tracking at all.
65    factory: Arc<dyn StorageFactory>,
66    /// Base configuration properties (endpoint, region, etc.), excluding credentials and
67    /// the two internal props consumed by [`RefreshableStorageFactory::build`].
68    base_props: HashMap<String, String>,
69    credentials_loader: Arc<dyn StorageCredentialsLoader>,
70    location: String,
71    table_ident: TableIdent,
72    /// Serializes calls to `credentials_loader` so concurrent failures caused by the same
73    /// stale credentials trigger exactly one refresh, not a thundering herd.
74    refresh_lock: tokio::sync::Mutex<()>,
75}
76
77impl State {
78    fn current(&self) -> (Arc<dyn Storage>, u64) {
79        let guard = self.current.read().expect("current lock poisoned");
80        (guard.0.clone(), guard.1)
81    }
82
83    /// Refreshes credentials and rebuilds the backend, unless another caller already did
84    /// so since `seen_version` was read (double-checked locking: cheap fast path with no
85    /// lock contention when a refresh already happened, `refresh_lock` serializes the
86    /// actual loader call when one is genuinely needed).
87    async fn refresh(&self, seen_version: u64) -> Result<()> {
88        if self.current.read().expect("current lock poisoned").1 > seen_version {
89            return Ok(());
90        }
91
92        let _guard = self.refresh_lock.lock().await;
93
94        if self.current.read().expect("current lock poisoned").1 > seen_version {
95            return Ok(());
96        }
97
98        let new_creds = self
99            .credentials_loader
100            .load_credentials(&self.table_ident, &self.location)
101            .await?;
102        let mut props = self.base_props.clone();
103        props.extend(new_creds.config);
104        let new_backend = self.factory.build(&StorageConfig::from_props(props))?;
105
106        let mut guard = self.current.write().expect("current lock poisoned");
107        let new_version = guard.1 + 1;
108        *guard = (new_backend, new_version);
109        Ok(())
110    }
111}

113/// Wraps the [`FileRead`] a backend returns from `reader()` so that *every*
114/// subsequent `.read()` call gets the same refresh-and-retry-once protection as a
115/// direct [`RefreshableStorage`] method call — not just the call that constructed it.
116///
117/// This matters because a reader is typically obtained once and then used for many
118/// range reads over the lifetime of a long read (e.g. a Parquet scan reading many row
119/// groups) — exactly the scenario a credential expiring mid-flight needs to survive,
120/// and exactly what retrying only the initial `reader()` construction would miss.
121///
122/// The current inner reader is held across calls behind a `tokio::sync::RwLock`
123/// rather than rebuilt on every `.read()` — every `Storage` method call rebuilds its
124/// backend's whole `opendal::Operator` from config, so re-resolving *that* on every
125/// single range read (as an earlier version of this wrapper did) would multiply that
126/// cost by the number of reads in a long scan instead of paying it once. Concurrent
127/// reads (e.g. parallel row-group fetches against the same file) share a read lock
128/// and aren't serialized on the happy path; only a failure-triggered rebuild briefly
129/// takes the write lock to swap in a freshly-built reader. This mirrors this crate's
130/// predecessor's own `RetryableReader` design exactly, including its reasoning.
131struct RefreshableReader {
132    state: Arc<State>,
133    path: String,
134    inner: tokio::sync::RwLock<(Box<dyn FileRead>, u64)>,
135}
136
137#[async_trait]
138impl FileRead for RefreshableReader {
139    async fn read(&self, range: std::ops::Range<u64>) -> Result<Bytes> {
140        let attempt = { self.inner.read().await.0.read(range.clone()).await };
141
142        match attempt {
143            Ok(v) => Ok(v),
144            Err(original_err) => {
145                let seen_version = self.inner.read().await.1;
146                self.state.refresh(seen_version).await?;
147                let (backend, new_version) = self.state.current();
148                let new_inner = backend
149                    .reader(&self.path)
150                    .await
151                    .map_err(|e| e.with_context("original_error", original_err.to_string()))?;
152                let result = new_inner.read(range).await;
153                *self.inner.write().await = (new_inner, new_version);
154                result.map_err(|e| e.with_context("original_error", original_err.to_string()))
155            }
156        }
157    }
158}

160/// A [`Storage`] that wraps another [`Storage`] (by default, any backend
161/// [`iceberg-storage-opendal`](iceberg_storage_opendal) can build) and transparently
162/// refreshes credentials on failure.
163///
164/// # Retry behavior
165///
166/// Every call is retried **at most once**: on any `Err`, credentials are refreshed and
167/// the backend rebuilt, then the same call is retried against the fresh backend — whether
168/// that retry succeeds or fails, its result is returned as-is (no further retries, no
169/// backoff). This is deliberate, not a simplification of a more precise design:
170///
171/// - The wrapped backend's errors don't reliably distinguish "credentials are stale" from
172///   any other failure. `iceberg-storage-opendal` maps every `opendal::Error` to a single
173///   flat `ErrorKind::Unexpected` regardless of cause, and a non-opendal `Storage`
174///   implementation isn't obligated to preserve any particular error shape at all — so
175///   there is no reliable signal to key a narrower heuristic on at this layer.
176/// - Retrying on any error, unconditionally, is exact parity with this crate's
177///   predecessor (an opendal-accessor-level wrapper), which had no error-kind filter
178///   either.
179///
180/// `delete_stream` is one exception: the input `Stream` is consumed by the first
181/// attempt and can't be safely replayed, so it is never retried — a failure always
182/// propagates immediately, matching this crate's predecessor's already-accepted lack of
183/// retry for several stream-based operations.
184///
185/// `reader`/`writer` are handled specially, since both return a long-lived object
186/// rather than a single result:
187/// - `reader()`'s returned [`FileRead`] is wrapped ([`RefreshableReader`]) so every
188///   subsequent `.read()` call gets this same retry-once treatment, not just the call
189///   that constructed it — this crate's predecessor gave every read this protection
190///   for a reader's whole lifetime (via an opendal-level accessor wrapper), and a
191///   long-lived reader used for many range reads (e.g. a Parquet scan) is exactly the
192///   scenario this whole mechanism exists to protect.
193/// - `writer()`'s returned [`FileWrite`] is NOT wrapped this way — see its own doc
194///   comment for why a stateful writer can't safely retry a failed write against a
195///   freshly-constructed one.
196pub struct RefreshableStorage {
197    /// `None` only after deserialization, which cannot reconstruct live credential state
198    /// (the credentials loader and current backend are never serializable) — every method
199    /// returns an error in that case rather than panicking.
200    state: Option<Arc<State>>,
201}
202
203impl Clone for RefreshableStorage {
204    fn clone(&self) -> Self {
205        Self {
206            state: self.state.clone(),
207        }
208    }
209}

268/// Refreshes credentials (double-checked against `seen_version`) and retries `op`
269/// against the freshly-refreshed backend. If `op` also fails, `original_err` is
270/// attached as context on the final error — so a caller debugging the failure can see
271/// what the *first* attempt failed with too, not just whatever the retry hit, which
272/// may be an unrelated, more confusing symptom (e.g. credentials refresh successfully
273/// but the retry then 404s for a genuinely separate reason).
274async fn refresh_and_retry<T, F, Fut>(
275    state: &State,
276    seen_version: u64,
277    original_err: Error,
278    op: F,
279) -> Result<T>
280where
281    F: FnOnce(Arc<dyn Storage>) -> Fut,
282    Fut: Future<Output = Result<T>>,
283{
284    state.refresh(seen_version).await?;
285    let (backend, _) = state.current();
286    op(backend)
287        .await
288        .map_err(|e| e.with_context("original_error", original_err.to_string()))
289}

291#[async_trait]
292#[typetag::serde(name = "RefreshableStorage")]
293impl Storage for RefreshableStorage {
294    async fn exists(&self, path: &str) -> Result<bool> {
295        let state = self.state()?;
296        let (backend, version) = state.current();
297        match backend.exists(path).await {
298            Ok(v) => Ok(v),
299            Err(e) => {
300                refresh_and_retry(state, version, e, |b| async move { b.exists(path).await }).await
301            }
302        }
303    }
304
305    async fn metadata(&self, path: &str) -> Result<FileMetadata> {
306        let state = self.state()?;
307        let (backend, version) = state.current();
308        match backend.metadata(path).await {
309            Ok(v) => Ok(v),
310            Err(e) => {
311                refresh_and_retry(state, version, e, |b| async move { b.metadata(path).await })
312                    .await
313            }
314        }
315    }
316
317    async fn read(&self, path: &str) -> Result<Bytes> {
318        let state = self.state()?;
319        let (backend, version) = state.current();
320        match backend.read(path).await {
321            Ok(v) => Ok(v),
322            Err(e) => {
323                refresh_and_retry(state, version, e, |b| async move { b.read(path).await }).await
324            }
325        }
326    }
327
328    async fn reader(&self, path: &str) -> Result<Box<dyn FileRead>> {
329        let state = Arc::clone(self.state()?);
330        let (backend, version) = state.current();
331        let (inner, version) = match backend.reader(path).await {
332            Ok(r) => (r, version),
333            Err(e) => {
334                state.refresh(version).await?;
335                let (backend, version) = state.current();
336                let inner = backend
337                    .reader(path)
338                    .await
339                    .map_err(|e2| e2.with_context("original_error", e.to_string()))?;
340                (inner, version)
341            }
342        };
343        Ok(Box::new(RefreshableReader {
344            state,
345            path: path.to_string(),
346            inner: tokio::sync::RwLock::new((inner, version)),
347        }))
348    }
349
350    async fn write(&self, path: &str, bs: Bytes) -> Result<()> {
351        let state = self.state()?;
352        let (backend, version) = state.current();
353        match backend.write(path, bs.clone()).await {
354            Ok(v) => Ok(v),
355            Err(e) => {
356                refresh_and_retry(
357                    state,
358                    version,
359                    e,
360                    |b| async move { b.write(path, bs).await },
361                )
362                .await
363            }
364        }
365    }
366
367    /// Unlike `reader()`, the returned [`FileWrite`] is NOT wrapped with per-call
368    /// retry — only this construction call is retried. A writer is frequently
369    /// stateful (e.g. a multipart upload accumulating state across `.write()` calls),
370    /// so a write that fails partway through can't be safely retried by constructing
371    /// a fresh writer and replaying just the failed call: any bytes already
372    /// successfully written to the old, now-abandoned writer would be silently lost
373    /// rather than replayed, potentially producing truncated output. Reads don't
374    /// have this problem — a range read is naturally idempotent, so retrying it
375    /// against a freshly-constructed reader is always safe.
376    async fn writer(&self, path: &str) -> Result<Box<dyn FileWrite>> {
377        let state = self.state()?;
378        let (backend, version) = state.current();
379        match backend.writer(path).await {
380            Ok(v) => Ok(v),
381            Err(e) => {
382                refresh_and_retry(state, version, e, |b| async move { b.writer(path).await }).await
383            }
384        }
385    }
386
387    async fn delete(&self, path: &str) -> Result<()> {
388        let state = self.state()?;
389        let (backend, version) = state.current();
390        match backend.delete(path).await {
391            Ok(v) => Ok(v),
392            Err(e) => {
393                refresh_and_retry(state, version, e, |b| async move { b.delete(path).await }).await
394            }
395        }
396    }
397
398    async fn delete_prefix(&self, path: &str) -> Result<()> {
399        let state = self.state()?;
400        let (backend, version) = state.current();
401        match backend.delete_prefix(path).await {
402            Ok(v) => Ok(v),
403            Err(e) => {
404                refresh_and_retry(
405                    state,
406                    version,
407                    e,
408                    |b| async move { b.delete_prefix(path).await },
409                )
410                .await
411            }
412        }
413    }
414
415    /// Not retried — see the [`RefreshableStorage`] docs.
416    async fn delete_stream(&self, paths: BoxStream<'static, String>) -> Result<()> {
417        self.state()?.current().0.delete_stream(paths).await
418    }
419
420    fn new_input(&self, path: &str) -> Result<InputFile> {
421        self.state()?;
422        Ok(InputFile::new(Arc::new(self.clone()), path.to_string()))
423    }
424
425    fn new_output(&self, path: &str) -> Result<OutputFile> {
426        self.state()?;
427        Ok(OutputFile::new(Arc::new(self.clone()), path.to_string()))
428    }
429}

431/// A [`StorageFactory`] that builds a [`RefreshableStorage`].
432///
433/// Inject it at catalog construction time via `with_storage_factory`. At table-load time
434/// the catalog populates [`StorageConfig`] with the table identity and metadata location
435/// (`FileIOBuilder::with_table_ident`/`with_location`); `build()` reads those to give the
436/// credentials loader context on every refresh.
437///
438/// By default, credentials are refreshed against a backend built via
439/// [`OpenDalResolvingStorageFactory`] (auto-detecting S3/GCS/OSS/ADLS/etc. from each
440/// path), so one injected factory instance can serve tables across multiple storage
441/// backends. Use [`RefreshableStorageFactory::with_factory`] to scope it to a specific,
442/// pre-configured [`StorageFactory`] instead.
443///
444/// # Example
445///
446/// ```rust,no_run
447/// use std::sync::Arc;
448///
449/// use iceberg::io::StorageCredentialsLoader;
450/// use iceberg_storage_refreshable::RefreshableStorageFactory;
451///
452/// // Implement your own loader:
453/// // let loader: Arc<dyn StorageCredentialsLoader> = ...;
454/// // let factory = Arc::new(RefreshableStorageFactory::new(loader));
455/// // catalog_config.with_storage_factory(factory);
456/// ```
457#[derive(Debug, Serialize, Deserialize)]
458pub struct RefreshableStorageFactory {
459    /// `None` only after deserialization (field is skipped).
460    #[serde(skip)]
461    credentials_loader: Option<Arc<dyn StorageCredentialsLoader>>,
462    /// `None` means "use the default `OpenDalResolvingStorageFactory`", resolved lazily at
463    /// `build()` time rather than eagerly in `new()` so a plain `#[serde(skip)]` (with no
464    /// `Default` requirement on `Arc<dyn StorageFactory>`) suffices here too.
465    #[serde(skip)]
466    inner_factory: Option<Arc<dyn StorageFactory>>,
467}
468
469impl RefreshableStorageFactory {
470    /// Creates a new factory that refreshes credentials via `credentials_loader`, building
471    /// backends via the default [`OpenDalResolvingStorageFactory`].
472    pub fn new(credentials_loader: Arc<dyn StorageCredentialsLoader>) -> Self {
473        Self {
474            credentials_loader: Some(credentials_loader),
475            inner_factory: None,
476        }
477    }
478
479    /// Scopes this factory to build backends via `factory` instead of the default
480    /// [`OpenDalResolvingStorageFactory`] — e.g. a pre-configured
481    /// `iceberg_storage_opendal::OpenDalStorageFactory::S3 { .. }` for a caller that only
482    /// ever serves one backend and wants to avoid per-call scheme resolution.
483    pub fn with_factory(mut self, factory: Arc<dyn StorageFactory>) -> Self {
484        self.inner_factory = Some(factory);
485        self
486    }
487}
488
489#[typetag::serde]
490impl StorageFactory for RefreshableStorageFactory {
491    fn build(&self, config: &StorageConfig) -> Result<Arc<dyn Storage>> {
492        let loader = self.credentials_loader.as_ref().ok_or_else(|| {
493            Error::new(
494                ErrorKind::Unexpected,
495                "RefreshableStorageFactory: credentials loader unavailable after deserialization",
496            )
497        })?;
498
499        // Extract runtime context from props, stripping the internal keys so they don't
500        // leak into the wrapped backend's own configuration.
501        let mut props = config.props().clone();
502        let location = props.remove(PROP_METADATA_LOCATION).unwrap_or_default();
503        let table_ident = props
504            .remove(PROP_TABLE_IDENT)
505            .and_then(|s| serde_json::from_str::<TableIdent>(&s).ok())
506            .unwrap_or_else(|| {
507                TableIdent::new(
508                    NamespaceIdent::new("unknown".to_string()),
509                    "unknown".to_string(),
510                )
511            });
512
513        let inner_factory = self
514            .inner_factory
515            .clone()
516            .unwrap_or_else(|| Arc::new(OpenDalResolvingStorageFactory::new()));
517
518        let initial_backend = inner_factory.build(&StorageConfig::from_props(props.clone()))?;
519
520        Ok(Arc::new(RefreshableStorage::new(
521            initial_backend,
522            inner_factory,
523            props,
524            Arc::clone(loader),
525            location,
526            table_ident,
527        )))
528    }
529}

598    /// A [`FileRead`] whose `.read()` succeeds only once `version >= fail_until_version`
599    /// — fixed at construction time (mirrors the version the *backend* that built it was
600    /// built with), so a reader obtained before a refresh keeps failing on every `.read()`
601    /// call even after a later, separately-obtained backend would succeed. This is what
602    /// makes `reader_retries_individual_read_calls_not_just_construction` a meaningful
603    /// test of [`RefreshableReader`] specifically, not just of `reader()`'s construction.
604    struct FakeReader {
605        version: u64,
606        fail_until_version: u64,
607    }
608
609    #[async_trait]
610    impl FileRead for FakeReader {
611        async fn read(&self, _range: std::ops::Range<u64>) -> Result<Bytes> {
612            if self.version < self.fail_until_version {
613                Err(Error::new(ErrorKind::Unexpected, "stale credentials"))
614            } else {
615                Ok(Bytes::from_static(b"data"))
616            }
617        }
618    }

620    #[derive(Debug, Clone, Serialize, Deserialize)]
621    struct FakeStorageFactory {
622        fail_until_version: u64,
623        /// Shared across every `FakeStorage` this factory builds, so a test can
624        /// count how many times `FakeStorage::reader()` was called -- i.e. how many
625        /// times the (simulated) backend operator was rebuilt.
626        #[serde(skip)]
627        reader_calls: Arc<AtomicUsize>,
628    }
629
630    #[typetag::serde]
631    impl StorageFactory for FakeStorageFactory {
632        fn build(&self, config: &StorageConfig) -> Result<Arc<dyn Storage>> {
633            let version = config
634                .get("version")
635                .map(|v| v.parse().unwrap())
636                .unwrap_or(0);
637            Ok(Arc::new(FakeStorage {
638                version,
639                fail_until_version: self.fail_until_version,
640                reader_calls: Arc::clone(&self.reader_calls),
641            }))
642        }
643    }

645    /// Returns credentials whose `config` bumps `version` by one each call, and tracks how
646    /// many times it was invoked.
647    struct TrackingLoader {
648        call_count: AtomicUsize,
649    }
650
651    impl fmt::Debug for TrackingLoader {
652        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
653            f.debug_struct("TrackingLoader").finish()
654        }
655    }
656
657    #[async_trait]
658    impl StorageCredentialsLoader for TrackingLoader {
659        async fn load_credentials(
660            &self,
661            _table_ident: &TableIdent,
662            _location: &str,
663        ) -> Result<StorageCredential> {
664            let n = self.call_count.fetch_add(1, Ordering::SeqCst) + 1;
665            Ok(StorageCredential {
666                prefix: String::new(),
667                config: HashMap::from([("version".to_string(), n.to_string())]),
668            })
669        }
670    }
671
672    fn build(fail_until_version: u64, loader: Arc<TrackingLoader>) -> Arc<dyn Storage> {
673        let factory =
674            RefreshableStorageFactory::new(loader).with_factory(Arc::new(FakeStorageFactory {
675                fail_until_version,
676                reader_calls: Arc::new(AtomicUsize::new(0)),
677            }));
678        factory.build(&StorageConfig::new()).unwrap()
679    }

758    #[tokio::test]
759    async fn reader_retries_individual_read_calls_not_just_construction() {
760        let loader = Arc::new(TrackingLoader {
761            call_count: AtomicUsize::new(0),
762        });
763        let storage = build(1, Arc::clone(&loader));
764
765        // FakeStorage::reader() never itself fails (only .read() calls simulate
766        // failure), so constructing the reader must not trigger a refresh on its own
767        // -- even though the backend it's bound to is already stale at this point.
768        let reader = storage.reader("x").await.unwrap();
769        assert_eq!(loader.call_count.load(Ordering::SeqCst), 0);
770
771        // The first .read() call is what actually hits the (still-stale) backend and
772        // must itself trigger exactly one refresh-and-retry -- proving retry isn't
773        // limited to whatever call constructed the reader.
774        let data = reader.read(0..4).await.unwrap();
775        assert_eq!(data, Bytes::from_static(b"data"));
776        assert_eq!(loader.call_count.load(Ordering::SeqCst), 1);
777
778        // A second .read() call on the *same*, already-obtained reader must observe
779        // the already-refreshed state and need no further refresh.
780        reader.read(0..4).await.unwrap();
781        assert_eq!(loader.call_count.load(Ordering::SeqCst), 1);
782    }

881    #[tokio::test]
882    async fn reader_reuses_inner_reader_across_successful_reads() {
883        // Regression test: an earlier version of RefreshableReader constructed a
884        // fresh inner reader (and therefore rebuilt the whole backend operator) on
885        // *every* .read() call, which would multiply that cost by the number of
886        // reads in a long scan instead of paying it once, unlike this crate's
887        // predecessor. Successful reads must reuse the same inner reader.
888        let loader = Arc::new(TrackingLoader {
889            call_count: AtomicUsize::new(0),
890        });
891        let reader_calls = Arc::new(AtomicUsize::new(0));
892        let factory =
893            RefreshableStorageFactory::new(loader).with_factory(Arc::new(FakeStorageFactory {
894                fail_until_version: 0,
895                reader_calls: Arc::clone(&reader_calls),
896            }));
897        let storage = factory.build(&StorageConfig::new()).unwrap();
898
899        let reader = storage.reader("x").await.unwrap();
900        assert_eq!(
901            reader_calls.load(Ordering::SeqCst),
902            1,
903            "reader() itself builds the inner reader exactly once"
904        );
905
906        for _ in 0..5 {
907            reader.read(0..4).await.unwrap();
908        }
909        assert_eq!(
910            reader_calls.load(Ordering::SeqCst),
911            1,
912            "5 successful reads must reuse the same inner reader, not rebuild it each time"
913        );
914    }
