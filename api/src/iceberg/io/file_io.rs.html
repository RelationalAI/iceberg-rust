30/// FileIO implementation, used to manipulate files in underlying storage.
31///
32/// FileIO wraps a `dyn Storage` with lazy initialization via `StorageFactory`.
33/// The storage is created on first use and cached for subsequent operations.
34///
35/// # Note
36///
37/// All paths passed to `FileIO` must be absolute paths starting with the scheme string
38/// appropriate for the storage backend being used.
39///
40/// This crate provides native support for local filesystem (`file://`) and
41/// memory (`memory://`) storage. For extensive storage backend support (S3, GCS,
42/// OSS, Azure, etc.), use the
43/// [`iceberg-storage-opendal`](https://crates.io/crates/iceberg-storage-opendal) crate.
44///
45/// # Example
46///
47/// ```rust,ignore
48/// use iceberg::io::{FileIO, FileIOBuilder};
49/// use iceberg::io::{LocalFsStorageFactory, MemoryStorageFactory};
50/// use std::sync::Arc;
51///
52/// // Create FileIO with memory storage for testing
53/// let file_io = FileIO::new_with_memory();
54///
55/// // Create FileIO with local filesystem storage
56/// let file_io = FileIO::new_with_fs();
57///
58/// // Create FileIO with custom factory
59/// let file_io = FileIOBuilder::new(Arc::new(LocalFsStorageFactory))
60///     .with_prop("key", "value")
61///     .build();
62/// ```
63#[derive(Clone, Debug)]
64pub struct FileIO {
65    /// Storage configuration containing properties
66    config: StorageConfig,
67    /// Factory for creating storage instances
68    factory: Arc<dyn StorageFactory>,
69    /// Cached storage instance (lazily initialized)
70    storage: Arc<OnceLock<Arc<dyn Storage>>>,
71}

96    /// This is useful for testing scenarios where persistent storage is not needed.
97    pub fn new_with_memory() -> Self {

104
105    /// Create a new FileIO backed by local filesystem storage.
106    ///
107    /// This is useful for local development and testing with real files.
108    pub fn new_with_fs() -> Self {
109        Self {
110            config: StorageConfig::new(),
111            factory: Arc::new(LocalFsStorageFactory),
112            storage: Arc::new(OnceLock::new()),
113        }
114    }
115
116    /// Serializes all portable state of this `FileIO` into a byte vector.
117    ///
118    /// This includes the storage configuration and factory, but not the cached storage instance.
119    /// The storage cache is rebuilt lazily on first use after calling [`FileIO::deserialize_all`].
120    ///
121    /// The serialized representation is not a stable format and may change between crate versions.
122    /// Applications should not rely on it for long-term storage or exchange it between incompatible
123    /// versions of this crate.
124    ///
125    /// All storage configuration properties are included in the serialized representation. These
126    /// properties may contain credentials or other sensitive values, so the returned bytes must be
127    /// protected in transit and at rest by the application embedding this crate.
128    ///
129    /// Storage factories are serialized through [`typetag`](https://docs.rs/typetag). Third-party
130    /// factories must use `#[typetag::serde]` on their [`StorageFactory`] implementation.
131    pub fn serialize_all(&self) -> Result<Vec<u8>> {
132        Ok(serde_json::to_vec(&_serde::SerializableFileIO {
133            config: &self.config,
134            factory: &self.factory,
135        })?)
136    }
137
138    /// Deserializes a `FileIO` previously produced by [`FileIO::serialize_all`].
139    ///
140    /// The receiving binary must use a compatible crate version and link the concrete factory
141    /// implementation so it is registered with `typetag`. Backend-specific requirements are
142    /// documented by each storage factory implementation.
143    pub fn deserialize_all(bytes: &[u8]) -> Result<Self> {
144        let _serde::DeserializedFileIO { config, factory } = serde_json::from_slice(bytes)?;
145        Ok(Self {
146            config,
147            factory,
148            storage: Arc::new(OnceLock::new()),
149        })
150    }
151
152    /// Get the storage configuration.
153    pub fn config(&self) -> &StorageConfig {
154        &self.config
155    }
156
157    /// Get or create the storage instance.
158    ///
159    /// The factory is invoked on first access and the result is cached
160    /// for all subsequent operations.
161    fn get_storage(&self) -> Result<Arc<dyn Storage>> {
162        // Check if already initialized
163        if let Some(storage) = self.storage.get() {
164            return Ok(storage.clone());
165        }
166
167        // Build the storage
168        let storage = self.factory.build(&self.config)?;
169
170        // Try to set it (another thread might have set it first)
171        let _ = self.storage.set(storage.clone());
172
173        // Return whatever is in the cell (either ours or another thread's)
174        Ok(self.storage.get().unwrap().clone())
175    }
176
177    /// Deletes file.
178    ///
179    /// # Arguments
180    ///
181    /// * path: It should be *absolute* path starting with scheme string used to construct [`FileIO`].
182    pub async fn delete(&self, path: impl AsRef<str>) -> Result<()> {
183        self.get_storage()?.delete(path.as_ref()).await
184    }
185
186    /// Remove the path and all nested dirs and files recursively.
187    ///
188    /// # Arguments
189    ///
190    /// * path: It should be *absolute* path starting with scheme string used to construct [`FileIO`].
191    ///
192    /// # Behavior
193    ///
194    /// - If the path is a file or not exist, this function will be no-op.
195    /// - If the path is a empty directory, this function will remove the directory itself.
196    /// - If the path is a non-empty directory, this function will remove the directory and all nested files and directories.
197    pub async fn delete_prefix(&self, path: impl AsRef<str>) -> Result<()> {
198        self.get_storage()?.delete_prefix(path.as_ref()).await
199    }
200
201    /// Delete multiple files from a stream of paths.
202    ///
203    /// # Arguments
204    ///
205    /// * paths: A stream of absolute paths starting with the scheme string used to construct [`FileIO`].
206    pub async fn delete_stream(
207        &self,
208        paths: impl Stream<Item = String> + Send + 'static,
209    ) -> Result<()> {
210        self.get_storage()?.delete_stream(paths.boxed()).await
211    }
212
213    /// Check file exists.
214    ///
215    /// # Arguments
216    ///
217    /// * path: It should be *absolute* path starting with scheme string used to construct [`FileIO`].
218    pub async fn exists(&self, path: impl AsRef<str>) -> Result<bool> {
219        self.get_storage()?.exists(path.as_ref()).await
220    }
221
222    /// Creates input file.
223    ///
224    /// # Arguments
225    ///
226    /// * path: It should be *absolute* path starting with scheme string used to construct [`FileIO`].
227    pub fn new_input(&self, path: impl AsRef<str>) -> Result<InputFile> {
228        self.get_storage()?.new_input(path.as_ref())
229    }
230
231    /// Creates output file.
232    ///
233    /// # Arguments
234    ///
235    /// * path: It should be *absolute* path starting with scheme string used to construct [`FileIO`].
236    pub fn new_output(&self, path: impl AsRef<str>) -> Result<OutputFile> {
237        self.get_storage()?.new_output(path.as_ref())
238    }

241/// Builder for [`FileIO`].
242///
243/// The builder accepts an explicit `StorageFactory` and configuration properties.
244/// Storage is lazily initialized on first use.
245#[derive(Clone, Debug)]
246pub struct FileIOBuilder {
247    /// Factory for creating storage instances
248    factory: Arc<dyn StorageFactory>,
249    /// Storage configuration
250    config: StorageConfig,
251}
252
253impl FileIOBuilder {
254    /// Creates a new builder with the given storage factory.
255    pub fn new(factory: Arc<dyn StorageFactory>) -> Self {
256        Self {
257            factory,
258            config: StorageConfig::new(),
259        }
260    }
261
262    /// Add a configuration property.
263    pub fn with_prop(mut self, key: impl ToString, value: impl ToString) -> Self {
264        self.config = self.config.with_prop(key.to_string(), value.to_string());
265        self
266    }
267
268    /// Add multiple configuration properties.
269    pub fn with_props(
270        mut self,
271        args: impl IntoIterator<Item = (impl ToString, impl ToString)>,
272    ) -> Self {
273        self.config = self
274            .config
275            .with_props(args.into_iter().map(|e| (e.0.to_string(), e.1.to_string())));
276        self
277    }
278
279    /// Set the runtime table identity context.
280    ///
281    /// Serializes the `TableIdent` as JSON into a well-known prop key so that a
282    /// credential-refreshing `StorageFactory` implementation (e.g. a fork's own
283    /// vended-credentials wrapper) can recover it and pass it to its credential loader on
284    /// refresh. Other storage factories ignore this prop.
285    pub fn with_table_ident(mut self, table_ident: crate::catalog::TableIdent) -> Self {
286        let json =
287            serde_json::to_string(&table_ident).expect("TableIdent serialization is infallible");
288        self.config = self.config.with_prop(PROP_TABLE_IDENT, json);
289        self
290    }
291
292    /// Set the runtime metadata location context.
293    ///
294    /// Stores the location under a well-known prop key so that a credential-refreshing
295    /// `StorageFactory` implementation can pass it to its credential loader on refresh.
296    /// Other storage factories ignore this prop.
297    pub fn with_location(mut self, location: impl Into<String>) -> Self {
298        self.config = self
299            .config
300            .with_prop(PROP_METADATA_LOCATION, location.into());
301        self
302    }
303
304    /// Get the storage configuration.
305    pub fn config(&self) -> &StorageConfig {
306        &self.config
307    }
308
309    /// Builds [`FileIO`].
310    pub fn build(self) -> FileIO {
311        FileIO {
312            config: self.config,
313            factory: self.factory,
314            storage: Arc::new(OnceLock::new()),
315        }
316    }
317}

358        Self { storage, path }
359    }
360

362    pub fn location(&self) -> &str {
363        &self.path
364    }
365
366    /// Check if file exists.
367    pub async fn exists(&self) -> Result<bool> {
368        self.storage.exists(&self.path).await

370
371    /// Fetch and returns metadata of file.
372    pub async fn metadata(&self) -> Result<FileMetadata> {
373        self.storage.metadata(&self.path).await
374    }
375
376    /// Read and returns whole content of file.
377    ///
378    /// For continuous reading, use [`Self::reader`] instead.

380        self.storage.read(&self.path).await
381    }
382
383    /// Creates [`FileRead`] for continuous reading.
384    ///
385    /// For one-time reading, use [`Self::read`] instead.
386    pub async fn reader(&self) -> Result<Box<dyn FileRead>> {
387        self.storage.reader(&self.path).await
388    }

687            .write("roundtrip".into())
688            .await
689            .unwrap();
690        assert!(file_io.storage.get().is_some());
691
692        let serialized = file_io.serialize_all().unwrap();
693        let deserialized = FileIO::deserialize_all(&serialized).unwrap();
694        assert!(deserialized.storage.get().is_none());
695        assert_eq!(
696            deserialized.config().get("test-property"),
697            Some(&"test-value".to_string())
698        );
699        assert!(deserialized.exists(path).await.unwrap());
700        assert_eq!(
701            deserialized.new_input(path).unwrap().read().await.unwrap(),
702            Bytes::from("roundtrip")
703        );
704        assert!(deserialized.storage.get().is_some());
705    }
